# A judge classifies each new user turn and picks one tier. Tool
# continuations reuse that decision when the client sends a session id.
[routes.main]
id = "switchyard-desktop"
type = "llm_classifier"
mode = "custom"
classifier_target = "quick_fixes"
targets = ["architecture", "refactors", "quick_fixes", "docs", "long_context"]
default_target = "architecture"
classify_trigger = "user_turn"
recent_turn_window = 6
max_output_tokens = 512
tool_calling = true
reasoning = true
prompt = '''
Choose exactly one model role for the requested coding or knowledge-work turn.

- architecture: Cross-module design, hard debugging, ambiguous work.
- refactors: Multi-file renames, extracting modules, tightening interfaces.
- quick_fixes: Tiny localized edits, syntax errors, short completions.
- docs: READMEs, comments, changelog entries, docstrings.
- long_context: Whole-repo questions and large diffs that need a big window.

Prefer the least powerful role that can reliably finish the whole requested
turn. When evidence is insufficient or several difficult capabilities are
mixed, select architecture.

Return exactly one minified JSON object in this shape:
{"decision":{"target":"architecture"}}
'''
response_schema = '''
{
  "type": "object",
  "properties": {
    "decision": {
      "type": "object",
      "properties": {
        "target": {
          "type": "string",
          "enum": ["architecture", "refactors", "quick_fixes", "docs", "long_context"]
        }
      },
      "required": ["target"],
      "additionalProperties": false
    }
  },
  "required": ["decision"],
  "additionalProperties": false
}
'''
